# default, so a half-written file cannot break the feed.

# CAUTION on wholeSceneSkipStages: it is NOT a free-form tuning knob. Stages produce

# missing effect. Add an id only with a reason. See docs/whole-scene-status.md.


# =============================================================================
# THE WHOLE-SCENE ROUTE — this is the renderer that feeds the panel
# =============================================================================

wholeSceneRender     = 1     # master switch for the second SceneDrawSystem.Draw
wholeSceneToPanel    = 1     # deliver its image to the panel
wholeSceneBuildBuffers = 1   # build our own ScreenBuffers (required)
wholeSceneWidth      = 512
wholeSceneHeight     = 512

# Render period, ms between feed renders. NO FLOOR: 0 = render every engine frame
wholeSceneIntervalMs = 0

wholeSceneCamera        = 1  # render from the orbit camera, not the player's
wholeSceneCameraRebuild = 2  # 2 = bake the 1:1 aspect into the projection

# Our own DrawContextManager. This is what makes VolumeRenderingContext, RTGIContext,
# StochasticTransparencyContext and WaterContext resize against OUR resolution instead
# of fighting the player's — they all hang off it. CloudJob does not, which is why it
# needs skip id 26 instead.
wholeSceneOwnDrawContexts = 1

# Our own shadow cascades. 1 = adaptive, 2 = force all. Requires stage 3 to run; the
# code force-runs it and logs why if you try to skip it.
wholeSceneOwnShadows        = 1
wholeSceneCascadeCount      = 2
wholeSceneCascadeResolution = 512

# Rebuild PlanetEnvironmentGroup's setup CBs from the orbit camera. Fixes the
# detached-atmosphere bug (the atmosphere rendering at the player's aim angle rather
# than on the planet).
wholeScenePlanetEnv = 1

wholeSceneAAMode        = -1  # -1 = do not scope DRS.AAMode at all (see stage 19/20)
wholeSceneNativeScaling = 0
wholeSceneExposure      = 0   # inert while skip id 25 is on; the feed follows the
                              # player's live adaptation instead

# Ray tracing in the feed. 0 = leave RaytracingSettings completely alone.
# rtFlags is deliberately EMPTY: scoping any RTGISettings field rebuilds the job
# snapshot, which is an async PSO compile, 10x/sec. An empty value here means "no
# flags", not "missing" — the parser treats it as an empty list on purpose.
wholeSceneDisableRaytracing = 0
wholeSceneRtFlags =

# Stages suppressed inside our nested Draw only. Each id has a documented reason in
# src/RttProbe/RttPlugin.cs next to the table entry.
#
#   0,1   raytracing TLAS + prepare — world-space shared, camera-dependent
#   2     RenderEnvironmentProbe    — shared probe atlas
#   5     UpdateSurfels             — water surfels
#   7     ProcessParticles          — particle simulation state
#   9     ExecuteHBAO               — confirmed device removal
#   16    DrawUI                    — keeps the player's HUD out of the feed
#   19,20 FSR: 20 forces IsFSREnabledAndAllowed false (no state change); 19 stops us
#         re-preparing FSR at 512 and wiping the player's TAA history
#   21    RenderFlares              — we share the engine's FlaresContext, never
#                                     advance its occlusion readback
#   24    AtmosphereLUTJob.DoWork   — shared per-planet LUTs
#   25    ConstantExposure          — read-only override; stops us stamping a constant
#                                     into the player's eye-adaptation history
#   26    CloudJob.DoWork           — shared CloudAccumulateLightAlpha is disposed and
#                                     recreated at our half-resolution 20x/sec.
#                                     CONFIRMED device removal (DRED page fault in
#                                     CloudShading). Costs the feed volumetric clouds
#                                     only; planet atmospheres are unaffected.
wholeSceneSkipStages = 0,1,2,5,7,9,16,19,20,21,24,25,26

emissivity  = 10         # LCD material emissive multiplier (shared definition)

# Scope PostProcessSettings.Bloom off for our render only. BloomJob retains its
# cascade borrows on the SHARED SceneDrawSystem._bloomJob.
wholeSceneNoBloom = 0

# Far plane for the FEED VIEW only, metres. 0 = player's far plane. The render's
# cost is CPU submit (draw count), so this is the main perf lever. VeryFarClipping
# is untouched: planets and their atmospheres still render. Try 3000-5000.
wholeSceneFarClip = 2500

# A/B gate on the FinalLDR resize (the phantom-bleed fix). 1 = resize to feed size
# (default). 0 = leave it at swapchain size — re-tests the pre-fix behaviour.
wholeSceneLdrResize = 1

# START-OF-FRAME SUBMISSION — the targeted fix for the session drift.
# 1 = record our render in Draw's PREFIX, before the player's frame, so our GPU work
#     executes while the CPU is still recording theirs (the drift is induced GPU idle,
#     not workload: our true GPU cost is ~3ms of a ~30ms ours-frame).
# 0 = legacy end-of-frame position.
# Costs one frame of feed latency. Live A/B — no rebuild, no settle window.
wholeSceneSubmitEarly = 0
